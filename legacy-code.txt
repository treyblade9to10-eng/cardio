function toast(msg, kind = '') {
  const t = document.createElement('div');
  t.className = 'toast ' + kind; t.textContent = msg;
  document.getElementById('toasts').appendChild(t);

}

function isGlitchActive() {
  const d = new Date();
  if (d.getDay() === 3 && d.getHours() === 16 && d.getMinutes() < 30) {
    if (!glitchAnnounced) { toast("THE WEDNESDAY GLITCH IS ON. Cards may come out upside down!", 'gold'); glitchAnnounced = true; }
    return true;
  }
  return false;
}

function saveKey(name) { return 'cardio-save-' + name.toLowerCase(); }

}

function login() {

function initials(n) { const w = n.replace(/[^A-Za-z0-9 ]/g, '').split(' ').filter(Boolean); return (w.length > 1 ? w[0][0] + w[1][0] : n.slice(0, 2)).toUpperCase(); }

function createCardHTML(card, showButton = false, tradeMode = false) {

  if(tradeMode) btn = `<button class="cbtn stage" onclick="event.stopPropagation(); stageForTrade(${card.id})">Stage</button>`;
  return `<div class="card ${rarity} ${auraClass} ${glitchClass} ${thankful}"><span class="card-rar">${rarity}</span><div class="card-art">${thankful ? '' : initials(card.n)}</div><div class="card-name">${card.n}</div><span class="card-val">${fmt(card.v)} c</span>${btn}</div>`;
}

}

/* ---------- SHOP ---------- */
function buyPack(t) {
  if(userData.coins < packData[t].cost) return toast("Not enough coins!");
  userData.coins -= packData[t].cost;
  let pulls = [];
  let cardCount = (t === 'golden') ? 3 : 5;
  let gotGod = false, thankfulHit = null;
  let glitchNow = isGlitchActive();
  for(let i = 0; i < cardCount; i++) {
    let r = Math.random() * 100, acc = 0, sel;
    for(let c of packData[t].cards) { acc += c.ch; if(r <= acc) { sel = c; break; } }
    let nc = {...(sel || packData[t].cards[0]), id: Math.random()};
    if(glitchNow && Math.random() < 0.25) nc.trait = 'upside-down';
    if(nc.r.includes("god-tier") || nc.r.includes("cosmic")) gotGod = true;
    if(THANKFUL_GODS.includes(nc.n)) thankfulHit = nc;
    pulls.push(nc); userData.inventory.push(nc);
  }
  document.getElementById('pack-opening-results').innerHTML = `<div class="results-head">${t} pack</div>` +
    pulls.map((c, i) => `<div class="reveal" style="animation-delay:${i * 0.12}s">${createCardHTML(c)}</div>`).join("");
  if(gotGod) {
    document.getElementById('main-body').classList.add('apply-shake');
    setTimeout(() => {
      document.getElementById('main-body').classList.remove('apply-shake');
      if(thankfulHit) {
        document.getElementById('big-reveal-title').textContent = thankfulHit.n;
        document.getElementById('big-reveal-card').innerHTML = createCardHTML(thankfulHit);
        document.getElementById('big-reveal').classList.remove('hidden');
      } else toast("GOD TIER PULL!", 'gold');
    }, 1000);
  }
  updateUI();
}

/* ---------- AUCTION (rival bidders) ---------- */
function minRaise(bid) { return Math.max(10, Math.ceil(bid * 0.08)); }
function clearBotTimers() { botTimers.forEach(clearTimeout); botTimers = []; }

function generateAuction() {
  clearBotTimers();
  auctionItems = []; bidDrafts = {};
  const all = Object.values(packData).flatMap(p => p.cards).filter(c => c.n !== "PURE GOLD" && c.n !== "ANCIENT FOSSIL");
  for(let i = 0; i < 4; i++) {
    const c = all[Math.floor(Math.random() * all.length)];
    const names = [...BOT_NAMES].sort(() => Math.random() - 0.5).slice(0, 2 + Math.floor(Math.random() * 2));
    // Each rival has a secret max. Some stop early, some go past the card's sell value.
    const bots = names.map(n => ({ n, max: Math.floor(c.v * rand(0.6, 1.4)) + 10 }));
    auctionItems.push({ card: { n: c.n, r: c.r, v: c.v, ch: c.ch }, bid: Math.max(5, Math.floor(c.v * 0.5)), top: "Market", mine: 0, bots, feed: "Opening bid", wasMine: false });
  }
  renderAuction();
}

function botBid(item, vsYou) {
  const need = item.bid + minRaise(item.bid);
  const able = item.bots.filter(b => b.n !== item.top && b.max >= need);
  if(!able.length) return false;
  const b = able[Math.floor(Math.random() * able.length)];
  const jump = minRaise(item.bid) * (1 + Math.floor(Math.random() * 3));
  const amt = Math.min(b.max, item.bid + jump);
  if(vsYou) { userData.coins += item.mine; item.mine = 0; }
  item.bid = amt; item.top = b.n; item.feed = `${b.n} bid ${fmt(amt)} c`;
  return true;
}

function customBid(idx) {
  const item = auctionItems[idx];
  const amt = parseInt(document.getElementById(`bid-input-${idx}`).value);
  const need = item.bid + minRaise(item.bid);
  if(item.top === "YOU") return toast("You're already the top bidder.");
  if(!amt || amt < need) return toast(`Bid at least ${fmt(need)} c`);
  if(userData.coins < amt) return toast("Not enough coins!");
  userData.coins -= amt;
  item.bid = amt; item.top = "YOU"; item.mine = amt; item.wasMine = true; item.feed = `You bid ${fmt(amt)} c`;
  delete bidDrafts[idx];
  if(auctionTime < 10) auctionTime = 10; // late bids give everyone a few more seconds
  const round = auctionItems;
  botTimers.push(setTimeout(() => {
    if(round !== auctionItems || item.top !== "YOU") return;
    if(botBid(item, true)) toast(`Outbid on ${item.card.n}! ${item.top} bid ${fmt(item.bid)} c`);
    else item.feed = "Nobody topped your bid... yet";
    renderAuction(); updateUI();
  }, rand(900, 2600)));
  renderAuction(); updateUI();
}

function tickAuction() {
  if(auctionTime > 0) {
    auctionTime--;
    let changed = false;
    auctionItems.forEach(item => { if(item.top !== "YOU" && Math.random() < 0.12 && botBid(item, false)) changed = true; });
    if(changed) renderAuction();
  } else {
    clearBotTimers();
    auctionItems.forEach(item => {
      if(item.top === "YOU") { userData.inventory.push({ ...item.card, id: Math.random() }); toast(`You won ${item.card.n} for ${fmt(item.bid)} c!`, 'good'); }
      else if(item.wasMine) toast(`${item.top} won ${item.card.n}. Your coins were refunded.`);
    });
    auctionTime = 120; generateAuction(); updateUI();
  }
  const el = document.getElementById('auction-timer');
  el.innerText = `0${Math.floor(auctionTime / 60)}:${(auctionTime % 60).toString().padStart(2, '0')}`;
  el.classList.toggle('hot', auctionTime <= 10);
}

function renderAuction() {
  document.getElementById('auction-list').innerHTML = auctionItems.map((item, i) => {
    const need = item.bid + minRaise(item.bid);
    const draft = bidDrafts[i] && bidDrafts[i] >= need ? bidDrafts[i] : need;
    const you = item.top === "YOU";
    const cls = you ? 'winning-bid' : (item.wasMine ? 'outbid' : '');
    return `<div class="lot ${cls}">
      ${createCardHTML(item.card)}
      <div class="row"><span>Top bidder</span><span class="who-top ${you ? 'you' : ''}">${you ? 'YOU' : item.top}</span></div>
      <div class="row"><span>Price</span><b>${fmt(item.bid)} c</b></div>
      <div class="row"><span>Card value</span><span>${fmt(item.card.v)} c</span></div>
      <div class="feed">${item.feed}</div>
      <div class="bidrow">
        <input type="number" class="field" id="bid-input-${i}" value="${draft}" min="${need}" oninput="bidDrafts[${i}]=parseInt(this.value)||0" ${you ? 'disabled' : ''}>
        <button class="btn primary" onclick="customBid(${i})" ${you ? 'disabled style="opacity:.5"' : ''}>Bid</button>
      </div>
    </div>`;
  }).join('');
}

/* ---------- TRADE ---------- */
function stageForTrade(id) {
  if(stagedCards.length >= 5) return toast("Max 5 cards!");
  let idx = userData.inventory.findIndex(c => c.id === id);
  if(idx === -1) return;
  stagedCards.push(userData.inventory.splice(idx, 1)[0]); renderTrade();
}
function unstage(idx) { userData.inventory.push(stagedCards.splice(idx, 1)[0]); renderTrade(); }
function renderTrade() {
  document.getElementById('staged-cards-list').innerHTML = stagedCards.map((c, i) => `<div style="display:inline-block; cursor:pointer;" onclick="unstage(${i})">${createCardHTML(c)}</div>`).join('');
  document.getElementById('trade-inventory-list').innerHTML = userData.inventory.map(c => createCardHTML(c, false, true)).join("");
  save();
}
function generateTradeCode() {
  if(stagedCards.length === 0) return toast("Stage some cards first!");
  let code = btoa(unescape(encodeURIComponent(JSON.stringify({ sender: userData.name, cards: stagedCards }))));
  document.getElementById('output-trade-code').value = code; stagedCards = []; renderTrade();
  toast("Code made. Send it to your friend.", 'good');
}
function receiveTradeCode() {
  try {
    let packageData = JSON.parse(decodeURIComponent(escape(atob(document.getElementById('input-trade-code').value.trim()))));
    packageData.cards.forEach(c => { c.id = Math.random(); userData.inventory.push(c); });
    document.getElementById('input-trade-code').value = ""; updateUI(); renderTrade();
    toast(`Got ${packageData.cards.length} card(s) from ${packageData.sender || 'a friend'}!`, 'good');
  } catch(e) { toast("Invalid Code!"); }
}

/* ---------- INVENTORY ---------- */
function sortInventoryByValue() { userData.inventory.sort((a, b) => b.v - a.v); renderCollection(); }
function bulkSell(r) {
  let v = 0, n = 0;
  userData.inventory = userData.inventory.filter(c => { if(c.r.split(' ')[0] === r) { v += c.v; n++; return false; } return true; });
  userData.coins += v; updateUI(); renderCollection();
  if(n) toast(`Sold ${n} ${r} card(s) for ${fmt(v)} c`, 'gold');
}
function renderCollection() {
  const query = document.getElementById('inventory-search').value.toLowerCase();
  const filtered = userData.inventory.filter(c => c.n.toLowerCase().includes(query));
  const total = userData.inventory.reduce((s, c) => s + c.v, 0);
  document.getElementById('inv-summary').textContent = `${userData.inventory.length} cards  |  worth ${fmt(total)} c`;
  document.getElementById('collection-list').innerHTML = filtered.length ? filtered.map(c => createCardHTML(c, true)).join("") : '<p class="sub">No cards yet. Hit the shop.</p>';
}
function sell(id) { let i = userData.inventory.findIndex(c => c.id == id); if(i === -1) return; userData.coins += userData.inventory[i].v; userData.inventory.splice(i, 1); renderCollection(); updateUI(); }

function updateUI() { document.getElementById('coin-count').innerText = fmt(userData.coins); save(); }
function showTab(t) {
  document.querySelectorAll('.tab-content').forEach(e => e.classList.add('hidden'));
  document.getElementById(t + '-tab').classList.remove('hidden');
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.toggle('active', b.dataset.tab === t));
  if(t === 'collection') renderCollection();
  if(t === 'trade') renderTrade();
  if(t === 'rankings') renderRankings();
  if(t === 'odds') renderOdds();
  if(t === 'auction') renderAuction();
}

}
